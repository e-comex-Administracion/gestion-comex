⚙️ Conexión Google Sheets

Para conectar la app con tu Google Sheet necesitás configurar un Apps Script. Seguí los pasos debajo.

📋 Instrucciones de configuración para escritura

  1. Abrí tu Google Sheet → Extensiones → Apps Script
  2. Borrá el contenido y pegá el script que se encuentra debajo de este panel
  3. Hacé clic en Implementar → Nueva implementación
  4. Tipo: Aplicación web → Ejecutar como: Yo → Acceso: Cualquier persona
  5. Copiá la URL generada y pegala en el campo de arriba
  6. Guardá la configuración y probá la conexión

📜 Apps Script (copiar y pegar en tu Google Sheet)

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(data.sheet);
  var result = {};

  if (data.action === 'append') {
    sheet.appendRow(data.row);
    result = {status:'ok', message:'Fila agregada'};
  } else if (data.action === 'update') {
    var range = sheet.getRange(data.row_index + 1, 1, 1, data.row.length);
    range.setValues([data.row]);
    result = {status:'ok', message:'Fila actualizada'};
  } else if (data.action === 'delete') {
    sheet.deleteRow(data.row_index + 1);
    result = {status:'ok', message:'Fila eliminada'};
  }

  return ContentService
    .createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var action = e && e.parameter && e.parameter.action;
  var callback = e && e.parameter && e.parameter.callback;

  if (action === 'getAll') {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheets = ['TIPOS_GASTO','PROVEEDORES','CLIENTES','OPERACIONES',
                  'FACTURAS_PROVEEDORES','FACTURAS_EMITIDAS','PAGOS','COBRANZAS','TIPO_CAMBIO'];
    var result = {status:'ok', data:{}};
    sheets.forEach(function(name) {
      var sh = ss.getSheetByName(name);
      if (!sh) return;
      var vals = sh.getDataRange().getValues();
      if (vals.length < 2) { result.data[name] = []; return; }
      var headers = vals[0];
      result.data[name] = vals.slice(1).map(function(row) {
        var obj = {};
        headers.forEach(function(h,i){ obj[h] = row[i] !== undefined ? String(row[i]) : ''; });
        return obj;
      });
    });
    var json = JSON.stringify(result);
    // JSONP: si viene callback, lo envuelve para que funcione desde file://
    if (callback) {
      return ContentService
        .createTextOutput(callback + '(' + json + ')')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService
      .createTextOutput(json)
      .setMimeType(ContentService.MimeType.JSON);
  }

  return ContentService
    .createTextOutput(callback ? callback + '({status:"ok"})' : '{"status":"ok"}')
    .setMimeType(callback ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON);
}
